cmake_minimum_required(VERSION 3.13)

include(pico_sdk_import.cmake)

project(test_blink C CXX ASM)

pico_sdk_init()

add_executable(test_blink
    test-minimal-blink.c
)

target_link_libraries(test_blink pico_stdlib)

# NO USB - this is the test
pico_enable_stdio_usb(test_blink 0)
pico_enable_stdio_uart(test_blink 0)

# Generate UF2
pico_add_extra_outputs(test_blink)
set(PICOTOOL_FETCH_FROM_GIT OFF CACHE BOOL "" FORCE)
